\section{Báo cáo công việc Tuần 2: Lập trình và Nghiệm thu sản phẩm}

\subsection{Phân bổ chi tiết lịch trình công việc hàng ngày}

Tuần thứ hai tập trung vào việc hiện thực hóa thiết kế từ tuần 1 thành sản phẩm chạy thực tế:

\begin{enumerate}
    \item \textbf{Thứ Hai:} Cài đặt và cấu hình môi trường build hoàn chỉnh: MSYS2/UCRT64, thư viện OpenCV 4.x và SQLite3. Viết tệp \texttt{CMakeLists.txt} để tự động hóa quá trình biên dịch và kiểm tra build lần đầu thành công.

    \item \textbf{Thứ Ba:} Lập trình module \texttt{DatabaseManager} trong hai tệp \texttt{database.h} và \texttt{database.cpp}: mở kết nối SQLite3, tạo bảng \texttt{residents}, hiện thực các hàm CRUD (thêm, xóa, truy vấn cư dân), lưu ảnh khuôn mặt mẫu và tải dữ liệu huấn luyện.

    \item \textbf{Thứ Tư:} Lập trình các chức năng nhận diện khuôn mặt AI: hàm \texttt{enrollFace()} thu thập ảnh mẫu qua camera, hàm \texttt{trainRecognizer()} huấn luyện mô hình LBPH và lưu ra file \texttt{lbph\_model.yml}, hàm \texttt{isModelUpToDate()} kiểm tra tính hiệu lực của model cache để tránh huấn luyện lại không cần thiết.

    \item \textbf{Thứ Năm:} Lập trình máy trạng thái thang máy (Elevator State Machine) gồm 7 trạng thái: \texttt{IDLE}, \texttt{DOOR\_OPENING\_BEFORE}, \texttt{BOARDING}, \texttt{DOOR\_CLOSING\_BEFORE}, \texttt{MOVING}, \texttt{ARRIVED}, \texttt{DOOR\_CLOSING\_AFTER}. Song song đó lập trình hàm \texttt{drawElevatorUI()} vẽ giao diện mô phỏng thang máy bằng OpenCV.

    \item \textbf{Thứ Sáu:} Tích hợp toàn bộ module vào hàm \texttt{main()}, lập trình giao tiếp cổng COM3 qua Win32 API (9600 baud, 8N1), thực hiện kiểm thử đơn vị (Unit Test) và chạy Demo nghiệm thu sản phẩm trực tiếp trước Mentor tại VNPT-Media.
\end{enumerate}

\subsection{Chi tiết kỹ thuật từng ngày}

\subsubsection{Thứ Hai --- Cấu hình môi trường và CMakeLists.txt}

Tệp \texttt{CMakeLists.txt} được viết để tự động tìm kiếm và liên kết thư viện OpenCV và SQLite3:

\begin{lstlisting}[language=make, caption={Tệp CMakeLists.txt cấu hình build dự án}]
cmake_minimum_required(VERSION 3.10)
project(SmartElevatorCamera)

set(CMAKE_CXX_STANDARD 17)
set(OpenCV_DIR "C:/msys64/ucrt64/lib/cmake/opencv4")

find_package(OpenCV REQUIRED)
find_package(SQLite3 REQUIRED)

add_executable(SmartElevatorCamera main.cpp database.cpp)

target_link_libraries(SmartElevatorCamera
    ${OpenCV_LIBS}
    ${SQLite3_LIBRARIES}
)
\end{lstlisting}

\subsubsection{Thứ Ba --- Lập trình Module DatabaseManager}

Module \texttt{DatabaseManager} phụ trách toàn bộ thao tác với cơ sở dữ liệu SQLite3. Bảng \texttt{residents} được khởi tạo bằng câu lệnh SQL sau:

\begin{lstlisting}[language=C++, caption={Khởi tạo bảng cư dân trong SQLite3}]
const char* sql = R"(
    CREATE TABLE IF NOT EXISTS residents (
        id           INTEGER PRIMARY KEY AUTOINCREMENT,
        name         TEXT    NOT NULL UNIQUE,
        apartment    TEXT    NOT NULL,
        target_floor INTEGER NOT NULL
    );
)";
sqlite3_exec(db, sql, nullptr, nullptr, nullptr);
\end{lstlisting}

\subsubsection{Thứ Tư --- Lập trình nhận diện khuôn mặt AI (OpenCV LBPH)}

\textbf{Hàm enrollFace()} --- Thu thập ảnh khuôn mặt mẫu qua camera, mỗi cư dân cần 30 ảnh mẫu grayscale kích thước $100 \times 100$ pixel:

\begin{lstlisting}[language=C++, caption={Hàm enrollFace() thu thập ảnh mẫu}]
// Moi lan nhan phim 'S': cat vung mat, resize 100x100, luu vao disk
if ((key == 's' || key == 'S') && !faces.empty()) {
    Mat faceROI = gray(faces[0]);
    Mat resized;
    resize(faceROI, resized, Size(100, 100));
    db.saveFaceImage(resident.id, resized, sampleCount++);
}
// Du 30 mau thi ket thuc tu dong
if (sampleCount >= ENROLLMENT_SAMPLES) break;
\end{lstlisting}

\textbf{Hàm trainRecognizer()} --- Huấn luyện mô hình LBPH từ toàn bộ ảnh mẫu và lưu ra file \texttt{.yml} để tái sử dụng:

\begin{lstlisting}[language=C++, caption={Huấn luyện và lưu mô hình LBPH}]
FaceTrainingData data = db.loadAllFaceTrainingData();
auto recognizer = LBPHFaceRecognizer::create();
recognizer->train(data.images, data.labels);
// Luu model ra file .yml, lan sau khoi dong tai truc tiep
recognizer->save("lbph_model.yml");
\end{lstlisting}

\textbf{Hàm isModelUpToDate()} --- So sánh thời gian sửa đổi của file model với các ảnh mẫu. Nếu model còn mới hơn, bỏ qua bước huấn luyện lại:

\begin{lstlisting}[language=C++, caption={Kiểm tra model cache còn hiệu lực không}]
LARGE_INTEGER modelTime = getFileModTime("lbph_model.yml");
if (modelTime.QuadPart == 0) return false; // Chua co model

WIN32_FIND_DATAA ffd;
HANDLE hFind = FindFirstFileA("faces\\*\\*.jpg", &ffd);
do {
    LARGE_INTEGER imgTime = {0};
    imgTime.LowPart  = ffd.ftLastWriteTime.dwLowDateTime;
    imgTime.HighPart = (LONG)ffd.ftLastWriteTime.dwHighDateTime;
    if (imgTime.QuadPart > modelTime.QuadPart)
        return false; // Co anh moi -> huan luyen lai
} while (FindNextFileA(hFind, &ffd));
FindClose(hFind);
return true; // Model con hieu luc
\end{lstlisting}

\subsubsection{Thứ Năm --- Máy trạng thái thang máy và giao diện đồ họa}

Cấu trúc \texttt{Elevator} lưu toàn bộ trạng thái thang máy và được cập nhật mỗi 30ms trong vòng lặp chính:

\begin{lstlisting}[language=C++, caption={Cập nhật trạng thái ELEVATOR\_MOVING}]
case ELEVATOR_MOVING: {
    int floorDiff = abs(elev.targetFloor - elev.currentFloor);
    double totalTravelTime = floorDiff * 1500.0; // 1500ms/tang
    double progress = elapsed / totalTravelTime;
    if (progress >= 1.0) {
        elev.currentFloor = elev.targetFloor;
        elev.state = ELEVATOR_ARRIVED;
    } else {
        // Cap nhat vi tri cabin theo thoi gian thuc
        elev.currentVisualFloor = elev.currentFloor
            + progress * (elev.targetFloor - elev.currentFloor);
    }
    break;
}
\end{lstlisting}

Hàm \texttt{drawElevatorUI()} vẽ giao diện mô phỏng bằng OpenCV gồm: LED panel hiển thị tầng, thông tin cư dân, cabin với cửa trượt hai bên, trục tầng và ảnh khuôn mặt nhúng trong cabin.

\subsubsection{Thứ Sáu --- Tích hợp COM và kiểm thử nghiệm thu}

Giao tiếp phần cứng qua cổng COM3 được lập trình bằng Win32 API với cấu hình 9600 baud, 8N1, gửi 1 byte mã tầng mỗi khi nhận diện thành công. Cơ chế chống spam lệnh được tích hợp thông qua State Machine: lệnh COM chỉ gửi đúng một lần khi thang máy chuyển từ \texttt{IDLE} sang \texttt{DOOR\_OPENING\_BEFORE}:

\begin{lstlisting}[language=C++, caption={Gửi lệnh COM một lần duy nhất, chống spam}]
// Chi gui lenh khi thang may chuyen tu IDLE -> hoat dong
bool trigger = (foundResident && elevator.state == ELEVATOR_IDLE);
if (updateElevator(elevator, trigger, bestResident, bestFaceCrop)) {
    // Gui 1 byte ma tang (vi du: tang 5 -> ky tu '5' = 0x35)
    char floorChar = '0' + elevator.targetFloor;
    if (useSerial) {
        sendData(hSerial, floorChar);
    } else {
        cout << "[DEMO] Tang " << elevator.targetFloor << endl;
    }
}
\end{lstlisting}

\subsection{Kết quả đạt được cuối kỳ thực tập}

\begin{itemize}
    \item Hoàn thành dự án cá nhân đúng tiến độ 2 tuần; chương trình chạy ổn định và đáp ứng đầy đủ các yêu cầu chức năng đề ra.
    \item Hệ thống nhận diện khuôn mặt đạt độ chính xác trên $92\%$ trong điều kiện ánh sáng ổn định trong nhà, vận hành ở tốc độ xấp xỉ 33 FPS.
    \item Được Mentor tại VNPT-Media đánh giá cao về khả năng tự nghiên cứu công nghệ mới (OpenCV, SQLite3, Win32 API), tư duy thiết kế hệ thống OOP và kỹ năng debug con trỏ, bộ nhớ C++.
    \item Hiểu rõ quy trình phát triển phần mềm chuyên nghiệp (SDLC): từ phân tích yêu cầu, thiết kế, lập trình, kiểm thử đến bàn giao sản phẩm trong môi trường doanh nghiệp thực tế.
    \item Mã nguồn được tổ chức rõ ràng theo mô hình module hóa, có chú thích đầy đủ và dễ bảo trì, mở rộng.
\end{itemize}
